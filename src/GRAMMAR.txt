START -> KW_PROGRAM SEMI USESOPT BLOCK DOT

USESOPT -> X -> ‘’
USESOPT -> KW_USES ID USESOPT

BLOCK -> COMMENT_MULTIPLE_LINES BLOCK
BLOCK -> COMMENT_SINGLE_LINE BLOCK
BLOCK -> WHITESPACE BLOCK
BLOCK -> X -> ‘’
BLOCK -> VAR_DEFINITION BLOCK
VAR_DEFINITION -> KW_VAR VAR_LIST BLOCK
VAR_LIST -> ‘’
VAR_LIST -> ID COMMA VAR_LIST
VAR_LIST -> ID COLON DATA_TYPE SEMI VAR_LIST
VAR_LIST -> ARRAY_DEC VAR_LIST
ARRAY_DEC -> ID COLON ARRAY LBRACKET DOUBLE_DOT RBRACKET COMPOUND_TYPE DATA_TYPE SEMI
/*Problema com identificação de:
NUMBER_REAL 1.
DOT .
NUMBER_INTEGER 5
Deveria ser:
NUMBER_INTEGER 1
DOUBLE_DOT ..
NUMBER_INTEGER 5
*/

DATA_TYPE -> BYTE
DATA_TYPE -> SHORTINT
DATA_TYPE -> SMALLINT
DATA_TYPE -> WORD
DATA_TYPE -> INTEGER
DATA_TYPE -> CARDINAL
DATA_TYPE -> LONGINT
DATA_TYPE -> LONGWORD
DATA_TYPE -> INT64
DATA_TYPE -> QWORD
DATA_TYPE -> DOUBLE
DATA_TYPE -> BOOLEAN
DATA_TYPE -> BOOLEAN16
DATA_TYPE -> BOOLEAN32
DATA_TYPE -> BOOLEAN64
DATA_TYPE -> BYTEBOOL
DATA_TYPE -> WORDBOOL
DATA_TYPE -> LONGBool
DATA_TYPE -> QWORDBool
DATA_TYPE -> ANSICHAR
DATA_TYPE -> WIDECHAR

BLOCK -> FUNCTION_ALL BLOCK 
FUNCTION_ALL -> KW_FUNCTION ID LPAR FUNC_PARAM RPAR COLON DATA_TYPE SEMI FUNCTION_BODY
FUNCTION_BODY -> COMMENT_MULTIPLE_LINES FUNCTION_BODY
FUNCTION_BODY -> COMMENT_SINGLE_LINE FUNCTION_BODY
FUNCTION_BODY -> WHITESPACE FUNCTION_BODY
FUNCTION_BODY -> VAR_DEFINITION FUNCTION_BODY
// FUNCTION_BODY -> KW_BEGIN ... KW_END SEMI BLOCK

FUNC_PARAM -> ‘’
FUNC_PARAM -> COMMA FUNC_PARAM
FUNC_PARAM -> SEMI FUNC_PARAM
FUNC_PARAM -> ID COLON DATA_TYPE FUNC_PARAM
FUNC_PARAM -> ID COLON DATA_TYPE FUNC_PARAM
FUNC_PARAM -> ID COLON ARRAY COMPOUND_TYPE DATA_TYPE FUNC_PARAM
FUNC_PARAM -> ID COLON ARRAY COMPOUND_TYPE DATA_TYPE FUNC_PARAM
