// This grammar must express at least the following structures of Pascal language.
/*
Arithmetic operators: +, -, *, /, %, parentheses, etc.
• Comparison operators: <,<=,>,>=,==,!=, etc.
• Logical operators: ||, &&, !
• Assignments
• Flow controls: if, if-else, while. (Not required, but allowed: for, do-while, switch, goto, break, continue)
• Function calls, which can be part of expressions
• Function declaration (It is not necessary to support nested functions or function pointers)
• Function return
• Variable type declaration
• Variable declaration, array variables (vectors)
• Access to array/vector index
• Scope: { }, begin-end, etc.
*/

// general scope of code
START -> KW_PROGRAM ID SEMI USESOPT BLOCK DOT

// Uses Directive
USESOPT -> X -> ''
USESOPT -> KW_USES ID USESOPT

// blocks of code (All code goes here)
BLOCK -> X -> ''
BLOCK -> VAR_DEFINITION BLOCK

// Variable Declaration
VAR_DEFINITION -> KW_VAR VAR_LIST BLOCK

// Diferent types of variable declaration
VAR_LIST -> ''
VAR_LIST -> ID COMMA VAR_LIST
VAR_LIST -> ID COLON DATA_TYPE SEMI VAR_LIST
VAR_LIST -> ARRAY_DEC VAR_LIST
ARRAY_DEC -> ID COLON ARRAY LBRACKET DOUBLE_DOT RBRACKET COMPOUND_TYPE DATA_TYPE SEMI
VAR_LIST -> ID COLON COMPOSITE_TYPE SEMI VAR_LIST

// Variables Types
DATA_TYPE -> SIMPLE_TYPE
DATA_TYPE -> COMPOSITE_TYPE

COMPOSITE_TYPE -> ARRAY COMPOUND_TYPE DATA_TYPE

SIMPLE_TYPE -> BYTE
SIMPLE_TYPE -> SHORTINT
SIMPLE_TYPE -> SMALLINT
SIMPLE_TYPE -> WORD
SIMPLE_TYPE -> INTEGER
SIMPLE_TYPE -> CARDINAL
SIMPLE_TYPE -> LONGINT
SIMPLE_TYPE -> LONGWORD
SIMPLE_TYPE -> INT64
SIMPLE_TYPE -> QWORD
SIMPLE_TYPE -> DOUBLE
SIMPLE_TYPE -> BOOLEAN
SIMPLE_TYPE -> BOOLEAN16
SIMPLE_TYPE -> BOOLEAN32
SIMPLE_TYPE -> BOOLEAN64
SIMPLE_TYPE -> BYTEBOOL
SIMPLE_TYPE -> WORDBOOL
SIMPLE_TYPE -> LONGBool
SIMPLE_TYPE -> QWORDBool
SIMPLE_TYPE -> ANSICHAR
SIMPLE_TYPE -> WIDECHAR

// functions declaration 
/*Tests:
KW_FUNCTION ID LPAREN ID COLON INTEGER SEMI ID COLON INTEGER SEMI ID COLON ARRAY COMPOUND_TYPE INTEGER RPAREN COLON INTEGER SEMI
KW_FUNCTION ID COLON INTEGER SEMI
KW_FUNCTION ID LPAREN RPAREN COLON INTEGER SEMI
KW_FUNCTION ID LPAREN ID COMMA ID COLON INTEGER RPAREN COLON BOOLEAN SEMI
KW_FUNCTION ID LPAREN ID COMMA ID COLON INTEGER RPAREN COLON INTEGER SEMI
*/
BLOCK -> FUNCTION_ALL BLOCK 

FUNCTION_ALL -> KW_FUNCTION ID PARAMETERS COLON DATA_TYPE SEMI FUNCTION_BODY
PARAMETERS -> LPAREN FUNC_PARAM_EXISTS RPAREN
PARAMETERS -> LPAREN RPAREN
PARAMETERS -> ''
FUNCTION_BODY -> VAR_DEFINITION FUNCTION_BODY
//TODO: FUNCTION_BODY -> KW_BEGIN /*BODY*/... KW_END SEMI BLOCK

FUNC_PARAM_EXISTS -> FUNC_PARAM FUNC_PARAM_ALL
FUNC_PARAM_ALL -> SEMI FUNC_PARAM FUNC_PARAM_ALL
FUNC_PARAM_ALL -> ''

// function parameters with COLON
FUNC_PARAM -> FUNC_PARAM_COLON COLON DATA_TYPE
//FUNC_PARAM -> FUNC_PARAM_COLON COLON DATA_TYPE INITIALIZATION
FUNC_PARAM_COLON -> ID FUNC_PARAM_COLON_HELP
FUNC_PARAM_COLON_HELP -> ''
FUNC_PARAM_COLON_HELP -> COMMA ID FUNC_PARAM_COLON_HELP

//INITIALIZATION -> ASSIGN EXPRESSION

// function body or code
BODY -> ID ASSIGN EXPRESSION SEMI BODY 
//TODO: SEMI pode ser opcional antes do end
//TODO: Not necessary for simplicity BODY -> FUNCTION_CALL SEMI BODY

TODO: EXPRESSION
EXPRESSION -> LPAREN  RPAREN
EXPRESSION -> UNARY_OPERATORS
EXPRESSION -> OPERATIONS EXPRESSION

NUMBERS_TYPES -> NUMBER_BINARY
NUMBERS_TYPES -> NUMBER_EXPONENTIAL
NUMBERS_TYPES -> NUMBER_REAL
NUMBERS_TYPES -> NUMBER_INTEGER
NUMBERS_TYPES -> NUMBER_HEXADECIMAL
NUMBERS_TYPES -> NUMBER_OCTAL
NUMBERS_TYPES -> ID
NUMBERS_TYPES -> LPAREN NUMBERS_TYPES RPAREN

//Unary operators(Highest precedence):
UNARY_OPERATORS -> KW_NOT NUMBERS_TYPES
UNARY_OPERATORS -> KW_PLUS NUMBERS_TYPES
UNARY_OPERATORS -> KW_MINUS NUMBERS_TYPES
UNARY_OPERATORS -> LPAREN UNARY_OPERATORS RPAREN
//EXPONENTIAL_OPERATION -> NUMBERS_TYPES EXP_SYM NUMBERS_TYPES

//Operations to suport left and right parentesis
//OPERATIONS -> EXPONENTIAL_OPERATION
OPERATIONS -> MULTIPLICATION_OPERATION
OPERATIONS -> DIVISION_OPERATION
OPERATIONS -> MOD_OPERATION
OPERATIONS -> AND_OPERATION
OPERATIONS -> SUM_OPERATION
OPERATIONS -> MINUS_OPERATION
OPERATIONS -> OR_OPERATION
OPERATIONS -> NOT_EQUAL_OPERATION
OPERATIONS -> LESS_OPERATION
OPERATIONS -> GREATER_OPERATION
OPERATIONS -> LESS_EQUAL_OPERATION
OPERATIONS -> GREATER_EQUAL_OPERATION
OPERATIONS -> NOT_EQUAL_OPERATION
OPERATIONS -> EQUAL_OPERATION
OPERATIONS -> LPAREN OPERATIONS RPAREN

//Multiplying operators(Second highest precedence):
MULTIPLICATION_OPERATION -> NUMBERS_TYPES MULT NUMBERS_TYPES
DIVISION_OPERATION -> NUMBERS_TYPES DIVIDE NUMBERS_TYPES
DIVISION_OPERATION -> NUMBERS_TYPES KW_DIV NUMBERS_TYPES
MOD_OPERATION -> NUMBERS_TYPES KW_MOD NUMBERS_TYPES
AND_OPERATION -> NUMBERS_TYPES KW_AND NUMBERS_TYPES

//Adding operators(Third highest precedence):
SUM_OPERATION -> NUMBERS_TYPES PLUS NUMBERS_TYPES
MINUS_OPERATION -> NUMBERS_TYPES MINUS NUMBERS_TYPES
OR_OPERATION -> NUMBERS_TYPES KW_OR NUMBERS_TYPES
NOT_EQUAL_OPERATION -> NUMBERS_TYPES REVERSED_NE NUMBERS_TYPES

// Relational operators(Lowest precedence):
LESS_OPERATION -> NUMBERS_TYPES LT NUMBERS_TYPES
GREATER_OPERATION -> NUMBERS_TYPES GT NUMBERS_TYPES
LESS_EQUAL_OPERATION -> NUMBERS_TYPES LE NUMBERS_TYPES
GREATER_EQUAL_OPERATION -> NUMBERS_TYPES GE NUMBERS_TYPES
NOT_EQUAL_OPERATION -> NUMBERS_TYPES NE NUMBERS_TYPES
EQUAL_OPERATION -> NUMBERS_TYPES EQUAL NUMBERS_TYPES



// Main program
MAIN_BODY -> BODY
BLOCK -> KW_BEGIN MAIN_BODY KW_END DOT EOF
