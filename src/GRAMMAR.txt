// general scope of code
START -> KW_PROGRAM ID SEMI USESOPT BLOCK DOT

// Uses Directive
USESOPT -> X -> ‘’
USESOPT -> KW_USES ID USESOPT

// blocks of code (All code goes here)
BLOCK -> X -> ‘’
BLOCK -> VAR_DEFINITION BLOCK

// Variable Declaration
VAR_DEFINITION -> KW_VAR VAR_LIST BLOCK

// Diferent types of variable declaration
VAR_LIST -> ‘’
VAR_LIST -> ID COMMA VAR_LIST
VAR_LIST -> ID COLON DATA_TYPE SEMI VAR_LIST
VAR_LIST -> ARRAY_DEC VAR_LIST
ARRAY_DEC -> ID COLON ARRAY LBRACKET DOUBLE_DOT RBRACKET COMPOUND_TYPE DATA_TYPE SEMI

// Variables Types
DATA_TYPE -> BYTE
DATA_TYPE -> SHORTINT
DATA_TYPE -> SMALLINT
DATA_TYPE -> WORD
DATA_TYPE -> INTEGER
DATA_TYPE -> CARDINAL
DATA_TYPE -> LONGINT
DATA_TYPE -> LONGWORD
DATA_TYPE -> INT64
DATA_TYPE -> QWORD
DATA_TYPE -> DOUBLE
DATA_TYPE -> BOOLEAN
DATA_TYPE -> BOOLEAN16
DATA_TYPE -> BOOLEAN32
DATA_TYPE -> BOOLEAN64
DATA_TYPE -> BYTEBOOL
DATA_TYPE -> WORDBOOL
DATA_TYPE -> LONGBool
DATA_TYPE -> QWORDBool
DATA_TYPE -> ANSICHAR
DATA_TYPE -> WIDECHAR

// functions declaration
BLOCK -> FUNCTION_ALL BLOCK 

FUNCTION_ALL -> KW_FUNCTION ID PARAMETERS COLON DATA_TYPE SEMI FUNCTION_BODY
PARAMETERS -> LPAR FUNC_PARAM_EXISTS RPAR
PARAMETERS -> LPAR RPAR
PARAMETERS -> ‘’
FUNCTION_BODY -> VAR_DEFINITION FUNCTION_BODY
//TODO: FUNCTION_BODY -> KW_BEGIN /*BODY*/... KW_END SEMI BLOCK

FUNC_PARAM_EXISTS -> FUNC_PARAM FUNC_PARAM_ALL
FUNC_PARAM_ALL -> SEMI FUNC_PARAM FUNC_PARAM_ALL
FUNC_PARAM_ALL -> ‘’

// function parameters with COLON
FUNC_PARAM -> FUNC_PARAM_COLON COLON DATA_TYPE
//FUNC_PARAM -> FUNC_PARAM_COLON COLON DATA_TYPE ASSIGN INICIALIZACAO
FUNC_PARAM_COLON -> ID FUNC_PARAM_COLON_HELP
FUNC_PARAM_COLON_HELP -> ‘’
FUNC_PARAM_COLON_HELP -> COMMA ID FUNC_PARAM_COLON_HELP

//INICIALIZACAO ->

// function body or code
BODY -> ID ASSIGN EXPRESSION SEMI BODY 
//TODO: SEMI pode ser opcional antes do end
//TODO: Not necessary for simplicity BODY -> FUNCTION_CALL SEMI BODY

// Main program
MAIN_BODY -> BODY
